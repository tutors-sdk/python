# Step 02: Dictionary Mastery

## Objective

Build a word frequency counter that processes text and provides various statistics.

## Implementation

```python
def word_frequency(text):
    """Count the frequency of each word in the text."""
    words = text.lower().split()
    # Remove basic punctuation
    clean_words = [word.strip(".,!?;:'\"()") for word in words]
    clean_words = [w for w in clean_words if w]  # Remove empty strings

    freq = {}
    for word in clean_words:
        freq[word] = freq.get(word, 0) + 1
    return freq

def top_n_words(freq_dict, n=5):
    """Return the top N most frequent words."""
    sorted_words = sorted(freq_dict.items(), key=lambda x: x[1], reverse=True)
    return sorted_words[:n]

def word_stats(freq_dict):
    """Return statistics about the word frequencies."""
    total_words = sum(freq_dict.values())
    unique_words = len(freq_dict)
    most_common = max(freq_dict, key=freq_dict.get)
    least_common = min(freq_dict, key=freq_dict.get)
    return {
        "total_words": total_words,
        "unique_words": unique_words,
        "most_common": (most_common, freq_dict[most_common]),
        "least_common": (least_common, freq_dict[least_common]),
    }
```

## Test It

```python
text = """
Python is a great programming language. Python is easy to learn.
Many people use Python for data science. Python is also great
for web development and automation.
"""

freq = word_frequency(text)
print("Word frequencies:", freq)
print("\nTop 5 words:", top_n_words(freq, 5))
print("\nStats:", word_stats(freq))
```
