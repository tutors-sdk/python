# Step 04: Student Database

## Objective

Combine lists, dicts, sets, and tuples to build a mini student database.

## Requirements

Build a `StudentDB` class that stores student records and supports these operations:

1. `add_student(name, age, courses)` -- courses is a list of course names
2. `get_student(name)` -- return student info as a dict
3. `add_course(name, course)` -- add a course to a student
4. `get_students_in_course(course)` -- return all students taking a course
5. `get_common_courses(name1, name2)` -- courses both students share
6. `get_average_age()` -- average age of all students
7. `summary()` -- print a formatted summary

## Implementation

```python
class StudentDB:
    def __init__(self):
        self._students = {}  # name -> {"age": int, "courses": set}

    def add_student(self, name, age, courses=None):
        self._students[name] = {
            "age": age,
            "courses": set(courses) if courses else set()
        }

    def get_student(self, name):
        student = self._students.get(name)
        if student:
            return {"name": name, "age": student["age"],
                    "courses": sorted(student["courses"])}
        return None

    def add_course(self, name, course):
        if name in self._students:
            self._students[name]["courses"].add(course)

    def get_students_in_course(self, course):
        return [name for name, info in self._students.items()
                if course in info["courses"]]

    def get_common_courses(self, name1, name2):
        s1 = self._students.get(name1, {}).get("courses", set())
        s2 = self._students.get(name2, {}).get("courses", set())
        return s1 & s2

    def get_average_age(self):
        if not self._students:
            return 0
        ages = [info["age"] for info in self._students.values()]
        return sum(ages) / len(ages)

    def summary(self):
        print(f"Student Database ({len(self._students)} students)")
        print(f"Average age: {self.get_average_age():.1f}")
        print("-" * 40)
        for name, info in sorted(self._students.items()):
            courses = ", ".join(sorted(info["courses"])) or "None"
            print(f"  {name} (age {info['age']}): {courses}")
```

## Test It

```python
db = StudentDB()
db.add_student("Alice", 20, ["Python", "Math", "Physics"])
db.add_student("Bob", 22, ["Python", "Chemistry"])
db.add_student("Charlie", 21, ["Math", "Physics", "Chemistry"])
db.add_student("Diana", 19, ["Python", "Math"])

db.add_course("Bob", "Math")

db.summary()

print(f"\nStudents in Python: {db.get_students_in_course('Python')}")
print(f"Students in Math: {db.get_students_in_course('Math')}")
print(f"Common courses (Alice & Charlie): {db.get_common_courses('Alice', 'Charlie')}")
```
