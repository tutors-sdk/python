## Problem 1: Find Common and Unique Elements

```python
def analyze_lists(list1, list2):
    """Find common, unique-to-each, and all elements."""
    set1, set2 = set(list1), set(list2)
    return {
        "common": set1 & set2,
        "only_in_first": set1 - set2,
        "only_in_second": set2 - set1,
        "all_unique": set1 | set2,
    }

result = analyze_lists(
    ["python", "java", "javascript", "c++"],
    ["python", "go", "rust", "javascript"]
)
for key, value in result.items():
    print(f"  {key}: {sorted(value)}")
```

## Problem 2: Deduplication Preserving Order

```python
def deduplicate(items):
    """Remove duplicates while preserving order."""
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

data = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5]
print(f"Original: {data}")
print(f"Deduped:  {deduplicate(data)}")
```

## Problem 3: Spell Checker

```python
def spell_check(text, dictionary):
    """Find words not in the dictionary."""
    words = set(text.lower().split())
    dictionary = set(word.lower() for word in dictionary)
    misspelled = words - dictionary
    return misspelled

known_words = {"the", "cat", "sat", "on", "mat", "a", "dog", "ran", "fast"}
text = "the catt sat on a mat and tha dog ran fastt"
errors = spell_check(text, known_words)
print(f"Possible misspellings: {errors}")
```
